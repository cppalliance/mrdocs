# The mrdocs command-line tool. Depends on the mrdocs-core library. The
# command line is parsed straight from argv against the reflected config
# schema, so the tool has no generated sources of its own.
file(GLOB_RECURSE TOOL_SOURCES CONFIGURE_DEPENDS
        ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp)

add_executable(mrdocs ${TOOL_SOURCES})
mrdocs_target_plugin_host_compile_definitions(mrdocs)

# What a plugin or an extension links, installed as `mrdocs::plugin-api`.
add_library(mrdocs-plugin-api INTERFACE)
add_library(mrdocs::plugin-api ALIAS mrdocs-plugin-api)
set_target_properties(mrdocs-plugin-api PROPERTIES EXPORT_NAME plugin-api)
# The include directories and options of the public headers.
target_link_libraries(mrdocs-plugin-api INTERFACE mrdocs-core-headers)
# A plugin links MrDocs dynamically, to the shared core or to the host
# executable, so it imports the API.
target_compile_definitions(mrdocs-plugin-api INTERFACE MRDOCS_DYNAMIC_LINK)
# The standard library tag MrDocs was built with, which Plugin.hpp compares with
# the plugin's own at its first compile.
if (MRDOCS_STDLIB_TAG_DEFINITIONS)
    target_compile_definitions(mrdocs-plugin-api INTERFACE
        ${MRDOCS_STDLIB_TAG_DEFINITIONS})
endif ()
# And the library a plugin's calls into MrDocs resolve against.
if (MRDOCS_BUILD_SHARED)
    target_link_libraries(mrdocs-plugin-api INTERFACE mrdocs-core)
else ()
    # The core is inside this executable (the plugin host), which exports the
    # API, so the plugin links the executable and carries no second copy of the
    # core.
    target_link_libraries(mrdocs-plugin-api INTERFACE mrdocs)
endif ()
if (MRDOCS_CLANG)
    target_compile_options(mrdocs PRIVATE -Wno-covered-switch-default)
endif ()

# A shared core is installed in the library directory, which the installed tool
# finds from its own location.
if (MRDOCS_BUILD_SHARED AND NOT WIN32)
    if (APPLE)
        set(MRDOCS_TOOL_ORIGIN "@loader_path")
    else ()
        set(MRDOCS_TOOL_ORIGIN "$ORIGIN")
    endif ()
    set_target_properties(mrdocs PROPERTIES
        INSTALL_RPATH "${MRDOCS_TOOL_ORIGIN}/../${CMAKE_INSTALL_LIBDIR}")
endif ()

# The config schema header is generated by utils/codegen; order it first.
if (TARGET mrdocs-codegen)
    add_dependencies(mrdocs mrdocs-codegen)
endif ()

# Install the tool into the shared export set (finalized by mrdocs_install()).
if (MRDOCS_INSTALL)
    install(TARGETS mrdocs-plugin-api EXPORT mrdocs-targets)
    install(TARGETS mrdocs
            EXPORT mrdocs-targets
            LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
            ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
            RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
